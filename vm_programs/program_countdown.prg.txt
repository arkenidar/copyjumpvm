# countdown loop-count
# Loops N times using an 8-bit decrementing counter, then terminates.
# Each iteration outputs the current counter value as one byte (LSB-first),
# so the output bytes read N, N-1, ..., 1.

# === EMBEDDED CONSTANT: N = 10 ===
# Counter is 8 bits at addresses 2..9 (mem[2]=bit0 ... mem[9]=bit7), LSB-first.
# To change N, edit the init bits below.
#   N=10 = 00001010b -> bit0=0 bit1=1 bit2=0 bit3=1 bit4=0 bit5=0 bit6=0 bit7=0

m 2 0   # counter bit0 = 0
m 3 1   # counter bit1 = 1
m 4 0   # counter bit2 = 0
m 5 1   # counter bit3 = 1
m 6 0   # counter bit4 = 0
m 7 0   # counter bit5 = 0
m 8 0   # counter bit6 = 0
m 9 0   # counter bit7 = 0

l main_loop

# --- loop body: output current counter (8 bits -> 1 byte) ---
m out 2
m out 3
m out 4
m out 5
m out 6
m out 7
m out 8
m out 9

# --- decrement counter by 1 (ripple borrow) ---
j 2 d0_is_1
m 2 1
j d1
l d0_is_1
m 2 0
j decrement_done
l d1
j 3 d1_is_1
m 3 1
j d2
l d1_is_1
m 3 0
j decrement_done
l d2
j 4 d2_is_1
m 4 1
j d3
l d2_is_1
m 4 0
j decrement_done
l d3
j 5 d3_is_1
m 5 1
j d4
l d3_is_1
m 5 0
j decrement_done
l d4
j 6 d4_is_1
m 6 1
j d5
l d4_is_1
m 6 0
j decrement_done
l d5
j 7 d5_is_1
m 7 1
j d6
l d5_is_1
m 7 0
j decrement_done
l d6
j 8 d6_is_1
m 8 1
j d7
l d6_is_1
m 8 0
j decrement_done
l d7
j 9 d7_is_1
m 9 1
j decrement_done
l d7_is_1
m 9 0

l decrement_done

# --- if counter is zero, terminate; else loop again ---
j 2 nonzero
j 3 nonzero
j 4 nonzero
j 5 nonzero
j 6 nonzero
j 7 nonzero
j 8 nonzero
j 9 nonzero
j end
l nonzero
j main_loop
